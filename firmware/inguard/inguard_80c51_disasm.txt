; Hardware model used for names:
;   Serial port mode 0 (SCON=0) = 8-bit synchronous byte bus to gate array U210 (RXD=data, TXD=clock)
;   P0[3:0] = gate-array register (ADD0-3), P0.4 = XRXW (1=read), P2.3 = IGSTB strobe

;   P2.2 = XHC HI connect, active low (Q12). P2.4 XEXOR=EXT OUT, P2.5 SEND, P2.6 HOLD, P2.7 XETRG,
;   P1.0 LEVEL, P1.1 ADBSY, P1.2 INCMP, P1.3 UPTRG, P1.4 ENTRG, P1.5/6 XCOT0/1, P1.7 NRFT, INT1 = XD_OVLD_F
